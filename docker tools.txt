browser in browser...

docker run -d \
  --name=firefox-sandbox \
  --security-opt seccomp=unconfined \
  -e PUID=1000 \
  -e PGID=1000 \
  -e TZ=Africa/Johannesburg \
  -p 3000:3000 \
  -p 3001:3001 \
  --shm-size="1gb" \
  --restart unless-stopped \
  lscr.io/linuxserver/firefox:latest





kali in browser...

docker run -d \
  --name=kali-linux \
  --security-opt seccomp=unconfined \
  -e PUID=1000 \
  -e PGID=1000 \
  -e TZ=Africa/Johannesburg \
  -e SUBFOLDER=/ \
  -e TITLE="Kali Linux" \
  -p 3011:3000 \
  -p 3009:3001 \
  --device /dev/dri:/dev/dri \
  --shm-size="1gb" \
  --restart unless-stopped \
  lscr.io/linuxserver/kali-linux:latest
